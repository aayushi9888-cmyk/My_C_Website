2. enter data into a file in capital letters

Source Code

#include <stdio.h>
void main()
{
FILE *fp;
char name[30], data[100];
int i;
printf("Enter file name: ");
scanf("%s", name);
fp = fopen(name, "w");
printf("Enter data: ");
scanf(" %[\n]", data);
for(i = 0; data[i] != '\0'; i++)
{
fputc(toupper((unsigned char)data[i]), fp);
}
fclose(fp);
printf("Data written in capital letters.");
}

Output

Enter file name: abc.txt
Enter data: hello world
Data written in capital letters

3. enter data into a file until the user wantsto stop

Source Code

#include <stdio.h>
void main()
{
FILE *fp;
char name[30], data[100];
char ch;
printf("Enter file name: ");
scanf("%s", name);
fp = fopen(name, "w");
do
{
printf("Enter data: ");
scanf(" %[\n]", data);
fprintf(fp, "%s\n", data);
printf("Do you want to continue? (y/n): ");
scanf(" %c", &ch);
} while(ch == 'y' || ch == 'Y');
fclose(fp);
printf("Data saved successfully.");
}

Output

Enter file name: abc.txt
Enter data: Hello
Do you want to continue? (y/n) y
Enter data: How are you
Do you want to continue? (y/n) n
Data saved successfully

4. copy one file into another file

Source Code

#include <stdio.h>
void main()
{
FILE *fp1, *fp2;
char name1[30], name2[30];
char ch;
printf("Enter first file name: ");
scanf("%s", name1);
fp1 = fopen(name1, "r");
printf("Enter second file name: ");
scanf("%s", name2);
fp2 = fopen(name2, "w");
while((ch = fgetc(fp1)) != EOF)
{
fputc(ch, fp2);
}
fclose(fp1);
fclose(fp2);
printf("File copied successfully.");
}

Output

Enter first file name: abc.txt
Enter second file name: xyz.txt
File copied successfully.

{
