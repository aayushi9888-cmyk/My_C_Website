6. Swap numbers using Call by address

Source Code

#include <stdio.h>
void swap(int *a, int *b)
{
int temp;
temp = *a;
*a = *b;
*b = temp;
}
void main()
{
int x, y;
printf("Enter two numbers: ");
scanf("%d%d", &x, &y);
swap(&x, &y);
printf("After swapping: %d %d", x, y); 
}

Output

Enter two numbers: 5 10
After swapping: 10 5

7.Make function for find out power of number, enter the number and power from user

Source Code

#include <stdio.h>
int power(int n, int p)
{
int i, result = 1;

for(i = 1; i <= p; i++)
{
result = result * n;
}
return result;
}
void main()
{
int n, p, ans;
printf("Enter number and power: ");
scanf("%d%d", &n, &p);
ans = power(n, p);
printf("Result = %d", ans);
return 0;
}

Output

Enter number and power: 2 3
Result = 8
