1.Program to a pointer should point the array and through pointer we will fill all the location of the array by asking the value from user and print the array

Source Code

#include <stdio.h>
void main()
{
int arr[5], i;
int *ptr;
printf("Enter 5 array elements:");
for(i = 0; i < 5; i++)
{
scanf("%d", &arr[i]);
}
ptr = &arr[0];
printf("\nValues and Addresses:\n");
for(i = 0; i < 5; i++)
{
printf("Value = %d\n", *(ptr + i));
printf("Address = %p\n",*(ptr + i));
}

2. write a program in which we are having a 2-D array of 2X3, insert te value through pointer and print the value

Source Code

#include <stdio.h>
 void main()
{
int arr[2][3];
int *ptr;
int i, j;
ptr = &arr[0][0];
printf("Enter 6 elements:\n");
for(i = 0; i < 2; i++)
{
for(j = 0; j < 3; j++)
{
scanf("%d", (ptr + i*3 + j));
}
}
printf("\nElements of 2-D array:\n");
 for(i = 0; i < 2; i++)
{
for(j = 0; j < 3; j++)
{
printf("%d\t", *(ptr + i*3 + j));
}
printf("\n");
}
}

Output

Enter 6 elements:
10 20 30
40 50 60
Elements of 2-D array:
10    20    30
40    50    60
