1.write a program to declare a union and display its members

Source Code

#include <stdio.h>
union student
{
int roll;
float marks;
};
void main()
{
union student s;
s.roll = 101;
printf("Roll Number = %d\n", s.roll);
s.marks = 85.5;
printf("Marks = %.f", s.marks);
}

Output

Roll Number = 101
Marks = 85.5

2.write a program to input and display student details using union

Source Code

#include  <stdio.h>
union student
{
int roll;
float marks;
};
void main()
{
union student s;
printf("Enter roll number: ");
scanf("%d", &s.roll);
printf("Roll Number = %d\n", s.roll);
printf("Enter marks: ");
scanf("%f", &s.marks);
printf("Marks = "%f", s.marks);
}

Output

Enter roll number: 101
Roll Number = 101
Enter marks: 90
Marks = 90.0
